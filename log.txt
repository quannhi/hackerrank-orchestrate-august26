CATEGORY_REGEX = {

   ),
   "Legal & Compliance": re.compile(
       r"\b(contract|nda|terms of service|privacy policy|compliance|litigation|gdpr|liability)\b",
       re.IGNORECASE,
   ),
}

# Standard category list for Gemini fallback
CATEGORIES = list(CATEGORY_REGEX.keys()) + ["General Inquiry"]

# ==========================================
# TIER 2: GEMINI FALLBACK CLASSIFIER
# ==========================================
def classify_with_gemini(text: str) -> str:
   """Fallback classifier using Gemini when Regex misses."""
   prompt = f"Categorize the following text into strictly one of these categories: {', '.join(CATEGORIES)}.\n\nText:\n{text}"
   
   # We use Flash for maximum speed/low cost, and constrain output format
   response = client.models.generate_content(
       model="gemini-2.5-flash",
       contents=prompt,
       config=types.GenerateContentConfig(
           system_instruction="You are a precise document classifier. Respond ONLY with the category name, nothing else.",
           temperature=0.0,
       ),
   )
   return response.text.strip()

# ==========================================
# HYBRID RULE ENGINE ROUTER
# ==========================================
def classify_text(text: str) -> dict:
   # 1. Try Tier 1: Regex Engine
   for category, pattern in CATEGORY_REGEX.items():
       if pattern.search(text):
           return {
               "category": category,
               "source": "Regex (Tier 1)",
               "cost": "$0.00"
           }

   # 2. Fallback to Tier 2: Gemini Engine
   category = classify_with_gemini(text)
   return {
       "category": category,
       "source": "Gemini API (Tier 2)",
       "cost": "API Usage"
   }

SCAM_RE = re.compile(
   r"(otp.*(leak|share|verify|confirm|pending|batao)|"

   r"(account|workspace|access).*(blocked|restricted|expire|closure|lock)|"

   r"(forward this to (ten|10) people|"

   r"do not break the chain|"
   r"bhagwan sabka bhala kare.*share|"

PROMOTION_RE = re.compile(
   r"(\b(discount|off|sale|deal|voucher|cashback|coupon|limited period|limited time|shop now)\b|TRY\d+|% off)",
   re.IGNORECASE,
)

GREETING_RE = re.compile(
   r"^(\s*)(good morning|good evening|good afternoon|have a good day|have a great day|bonjour|hi|hello)(\s*)[\.\!\?]*$",
   re.IGNORECASE,
)

FORWARD_RE = re.compile(
   r"^(\s*)(fwd|forwarded):",

def classify_tier1_regex(message_text: str):
   """
   Fast regex scanner.
   Returns (action, message_type, reason) if confident, or (None, None, None) to hand off to Gemini.
   """
   if not isinstance(message_text, str) or not message_text.strip():
       return None, None, None

   text = message_text.strip()

   # 1. High-risk Scams / Phishing / Prompt Injection -> Mute
   if SCAM_RE.search(text):
       return (
           "mute",
           "scam",
           "High-risk security scam, phishing link, or prompt injection attempt.",
       )

   # 2. Chain messages / Spam forwarders -> Mute
   if SPAM_RE.search(text):
       return (
           "mute",
           "spam",
           "Unwanted chain letter, spam broadcast, or mass forward.",
       )

   # 3. Simple standalone greetings -> Digest
   if GREETING_RE.match(text):
       return (
           "digest",
           "greeting",
           "Harmless polite greeting with no immediate call to action.",
       )

   return None, None, None

import os
import json
from pydantic import BaseModel, Field
from enum import Enum
from google import genai
from google.genai import types

# ---------------------------------------------------------------------------
# 1. Define Strict Output Schema using Pydantic
# ---------------------------------------------------------------------------
class ActionEnum(str, Enum):
   notify = "notify"
   digest = "digest"
   mute = "mute"

class MsgTypeEnum(str, Enum):
   personal = "personal"
   urgent = "urgent"
   event = "event"
   payment = "payment"
   business_update = "business_update"
   promotion = "promotion"
   greeting = "greeting"
   forward = "forward"
   spam = "spam"
   scam = "scam"
   unknown = "unknown"

class RouterDecision(BaseModel):
   action: ActionEnum = Field(description="The routing action: notify, digest, or mute.")
   message_type: MsgTypeEnum = Field(description="The specific category of the message.")
   reason: str = Field(description="A 1-2 sentence explanation of why this decision was made, referencing context flags.")
   confidence: str = Field(description="High, medium, or low confidence in this classification.")

# ---------------------------------------------------------------------------
# 2. Main LLM Router Class
# ---------------------------------------------------------------------------
class LLMRouter:
   def __init__(self):
       # Automatically picks up GEMINI_API_KEY from the environment
       self.client = genai.Client()
       # Using flash for fast, cost-effective routing tasks
       self.model_name = "gemini-2.5-flash" 

   def construct_prompt(self, context: dict) -> str:
       """Translates the rich context dictionary into a clear prompt for Gemini."""
       
       # Format the historical metrics and flags for easy LLM reading
       flags = context.get('security_flags', {})
       metrics = context.get('metrics', {})
       
       prompt = f"""
       You are an intelligent notification router. Analyze the following incoming message and its context to decide its routing action and category.
       
       [ROUTING RULES]
       - ACTION 'mute': Use for spam, scams, heavy forwards, or if domain_mismatch is True.
       - ACTION 'notify': Use for urgent matters, personal direct messages, critical events, or requested payment updates.
       - ACTION 'digest': Use for promotions, standard business updates, greetings, or low-priority interactions.
       
       [MESSAGE DETAILS]
       - Text: "{context.get('message_text', '')}"
       - Conversation Type: {context.get('conversation_type')}
       - Forwarded Count: {context.get('forwarded_count')}
       
       [SECURITY & HISTORY CONTEXT]
       - Domain Mismatch (Scam Risk): {flags.get('domain_mismatch')}
       - Business Verified: {flags.get('is_business_verified')}
       - Prior User Reports: {flags.get('has_prior_user_reports')}
       - Prior User Mutes: {flags.get('has_prior_user_mutes')}
       - User Historical Reply Rate: {metrics.get('reply_rate', 0.0):.2f} (1.0 = always replies, 0.0 = never replies)
       
       Analyze the text, the provided context, and any attached media to classify this message accurately.
       """
       return prompt

   def analyze_message(self, context: dict) -> dict:
       """Sends the context and media to Gemini and returns the structured decision."""
       try:
           contents = [self.construct_prompt(context)]
           
           # Handle Media if it exists
           media_path = context.get("media_path")
           uploaded_file = None
           if media_path and os.path.exists(media_path):
               print(f"Uploading media for analysis: {media_path}")
               uploaded_file = self.client.files.upload(file=media_path)
               contents.append(uploaded_file)

           # Call Gemini with Structured Outputs
           response = self.client.models.generate_content(
               model=self.model_name,
               contents=contents,
               config=types.GenerateContentConfig(
                   response_mime_type="application/json",
                   response_schema=RouterDecision,
                   temperature=0.1, # Low temperature for more deterministic/logical outputs
               ),
           )
           
           # The SDK handles the JSON parsing into a dict when using response_schema
           decision = json.loads(response.text)
           
           # Cleanup uploaded file to save space in your Gemini workspace
           if uploaded_file:
               self.client.files.delete(name=uploaded_file.name)
               
           return decision

       except Exception as e:
           print(f"LLM Routing failed for message {context.get('message_id')}: {e}")
           # Safe Fallback
           return {
               "action": "digest",
               "message_type": "unknown",
               "reason": f"Fallback applied due to API error: {str(e)}",
               "confidence": "low"
           }

class LLMRouter:

   def construct_prompt(self, context: dict) -> str:

               "message_type": "unknown",

Create code/test_batch.py to run a slice of 10 to 20 messages at a time. It prints the exact DataFrame format and exports a temporary test_output.csv for quick inspection.
import os
import pandas as pd
from dotenv import load_dotenv

from context_builder import ContextBuilder
from rule_engine import classify_tier1_regex
from router_llm import LLMRouter

def run_batch_test(start_idx: int = 0, batch_size: int = 10):
   load_dotenv()
   
   base_dir = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
   dataset_dir = os.path.join(base_dir, "dataset")
   messages_path = os.path.join(dataset_dir, "messages.csv")

   messages_df = pd.read_csv(messages_path)
   
   # Slice only the requested batch
   end_idx = min(start_idx + batch_size, len(messages_df))
   batch_df = messages_df.iloc[start_idx:end_idx]
   
   print(f"--- RUNNING TEST BATCH ({start_idx} to {end_idx - 1}) ---")
   
   context_builder = ContextBuilder(dataset_dir=dataset_dir)
   llm_router = LLMRouter()
   
   results = []

   for idx, row in batch_df.iterrows():
       msg_id = row['message_id']
       msg_text = row.get('message_text', '')
       
       # Tier 1 Regex
       action, msg_type, reason = classify_tier1_regex(msg_text)
       
       if action is not None:
           ctx = context_builder.get_message_context(msg_id)
           raw_evidence = ctx.get("evidence_ids", [])
           evidence_str = ";".join(raw_evidence) if raw_evidence else "none"
           
           decision = {
               "message_id": msg_id,
               "action": action,
               "message_type": msg_type,
               "reason": reason,
               "confidence": 1.00,  # Float representation for regex
               "evidence_message_ids": evidence_str
           }
           print(f"[{idx}] {msg_id} -> TIER 1 | Action: {action} | Conf: 1.00 | Evidence: {evidence_str}")

       # Tier 2 LLM Router
       else:
           ctx = context_builder.get_message_context(msg_id)
           decision_raw = llm_router.analyze_message(ctx)
           
           # Format evidence IDs
           raw_evidence = ctx.get("evidence_ids", [])
           evidence_str = ";".join(raw_evidence) if raw_evidence else "none"
           
           # Ensure numerical float confidence
           conf_val = decision_raw.get("confidence", 0.85)
           try:
               conf_val = float(conf_val)
           except (ValueError, TypeError):
               conf_val = 0.85

           decision = {
               "message_id": msg_id,
               "action": decision_raw.get("action", "digest"),
               "message_type": decision_raw.get("message_type", "unknown"),
               "reason": decision_raw.get("reason", "Analyzed by LLM context router."),
               "confidence": round(conf_val, 2),
               "evidence_message_ids": evidence_str
           }
           print(f"[{idx}] {msg_id} -> TIER 2 | Action: {decision['action']} | Conf: {decision['confidence']} | Evidence: {evidence_str}")

       results.append(decision)

   out_df = pd.DataFrame(results)
   required_cols = ["message_id", "action", "message_type", "reason", "confidence", "evidence_message_ids"]
   out_df = out_df[required_cols]

   print("\n--- SAMPLE OUTPUT DATAFRAME ---")
   print(out_df.to_string(index=False))
   
   test_out_path = os.path.join(base_dir, "test_output.csv")
   out_df.to_csv(test_out_path, index=False)
   print(f"\nSaved test batch to {test_out_path}")

if __name__ == "__main__":
   # Change start_idx and batch_size as needed (e.g. 0 to 10, 10 to 20)
   run_batch_test(start_idx=0, batch_size=10)
